      entity?: string;
      userId?: number;
      action?: string;
      from?: Date;
      to?: Date;
    }

    export const AuditLogModelQuery = {
      findMany: (filter: AuditLogFilter) =>
        prisma.auditLog.findMany({
          where: {
            entity: filter.entity,
            userId: filter.userId,
            action: filter.action,
            createdAt: filter.from || filter.to ? { gte: filter.from, lte: filter.to } : undefined,

          orderBy: { createdAt: "desc" },
          include: { user: { select: { id: true, username: true } } },
        }),

    ```typescript
    import { z } from "zod";
    import { AuditLogModelQuery } from "../models/auditLog.model.js";
    import { asyncHandler } from "../utils/asyncHandler.js";

    const filterSchema = z.object({
      entity: z.string().optional(),
      userId: z.coerce.number().int().optional(),
      action: z.string().optional(),
      from: z.coerce.date().optional(),
      to: z.coerce.date().optional(),
    });

    export const listAuditLogs = asyncHandler(async (req, res) => {
      const filter = filterSchema.parse(req.query);
      res.json(await AuditLogModelQuery.findMany(filter));
    });
    ```

## Trust Boundaries

| Boundary | Description |
|----------|--------------|
| client -> GET /api/audit-logs | The audit trail itself is a sensitive resource — read access must be restricted, and write access must be structurally impossible |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-13 | Tampering | Audit log entries edited/deleted to hide unauthorized actions | mitigate | No PUT/DELETE route is ever registered on `auditLogRoutes` — a deliberate architectural choice (not a DB-level immutability guarantee, since MySQL has no native row-immutability feature at this project's scale); documented as an accepted trust boundary, proven by `audit.immutable.test.ts`'s 404 assertions |
| T-02-14 | Information Disclosure | IDOR via `?userId=X` allowing a lower-privilege user to view another user's actions | accept | Per D-04, the entire endpoint is gated to `SYSTEM_ADMIN`/`MANAGER_APPROVER` only — no role below Manager can reach the endpoint at all, so the `userId` filter parameter carries no additional exposure beyond what those two roles already have access to |
| T-02-15 | Denial of Service | Unbounded `GET /api/audit-logs` with no filter returning the entire table as it grows | accept | Out of scope for this phase per RESEARCH.md; acceptable at current data volume, filtering by date range is available to narrow results, and pagination can be added in a future phase without breaking this endpoint's contract |
</threat_model>
