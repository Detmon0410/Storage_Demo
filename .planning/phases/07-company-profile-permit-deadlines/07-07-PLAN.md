    ```typescript
    const permitBuckets = useMemo(() => {
      const buckets: Record<string, number> = { PREPARATION: 0, NOTIFY: 0, WARNING: 0, IMPORTANT_WARNING: 0, EXPIRED: 0 };
      for (const l of licenses) {
        if (l.status in buckets) buckets[l.status] += 1;
      }
      return buckets;
    }, [licenses]);
    ```
